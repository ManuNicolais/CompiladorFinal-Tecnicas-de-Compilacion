// Ejemplo con FOR, BREAK y CONTINUE

int buscarNumero(int arreglo[10], int objetivo) {
    int i;
    int encontrado;
    encontrado = 0;
    
    for (i = 0; i < 10; i = i + 1) {
        if (arreglo[i] == objetivo) {
            encontrado = 1;
            break;
        }
    }
    
    return encontrado;
}

int sumarPares(int valores[5]) {
    int suma;
    int i;
    suma = 0;
    
    for (i = 0; i < 5; i = i + 1) {
        if (valores[i] % 2 != 0) {
            continue;
        }
        suma = suma + valores[i];
    }
    
    return suma;
}

int main() {
    int numeros[5];
    int contador;
    int resultado;
    
    // Inicializar array
    numeros[0] = 2;
    numeros[1] = 4;
    numeros[2] = 6;
    numeros[3] = 8;
    numeros[4] = 10;
    
    // Contar pares
    contador = 0;
    for (contador = 0; contador < 5; contador = contador + 1) {
        resultado = numeros[contador];
    }
    
    return 0;
}
