// Ejemplo con expresiones complejas y optimización

int calcularPolinomio(int x) {
    int resultado;
    int temp;
    
    // 3x^2 + 2x + 5
    resultado = 3 * x * x;
    temp = 2 * x;
    resultado = resultado + temp;
    resultado = resultado + 5;
    
    return resultado;
}

double temperatura(double celsius) {
    double fahrenheit;
    fahrenheit = celsius * 9.0 / 5.0;
    fahrenheit = fahrenheit + 32.0;
    return fahrenheit;
}

int fibonacci(int n) {
    int a;
    int b;
    int temp;
    int i;
    
    a = 0;
    b = 1;
    
    i = 0;
    while (i < n) {
        temp = a + b;
        a = b;
        b = temp;
        i = i + 1;
    }
    
    return a;
}
